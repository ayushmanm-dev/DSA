import java.util.LinkedList;
import java.util.Queue;

class Solution {
    public int minMoves(String[] classroom, int energy) {
        int m = classroom.length;
        int n = classroom[0].length();
        int sx = -1, sy = -1;
        int numLitter = 0;
        int[][] litterIdx = new int[m][n];
        
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                litterIdx[i][j] = -1;
                char c = classroom[i].charAt(j);
                if (c == 'S') {
                    sx = i;
                    sy = j;
                } else if (c == 'L') {
                    litterIdx[i][j] = numLitter++;
                }
            }
        }
        
        int fullMask = (1 << numLitter) - 1;
        if (fullMask == 0) return 0; 
        
        int[][][] bestEnergy = new int[m][n][1 << numLitter];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < (1 << numLitter); k++) {
                    bestEnergy[i][j][k] = -1;
                }
            }
        }
        
        Queue<int[]> q = new LinkedList<>();
        q.add(new int[]{sx, sy, 0, energy, 0});
        bestEnergy[sx][sy][0] = energy;
        
        int[] dirX = {-1, 1, 0, 0};
        int[] dirY = {0, 0, -1, 1};
        
        while (!q.isEmpty()) {
            int[] curr = q.poll();
            int x = curr[0];
            int y = curr[1];
            int mask = curr[2];
            int e = curr[3];
            int steps = curr[4];
            
            if (e == 0 && classroom[x].charAt(y) != 'R') {
                continue;
            }
            
            for (int d = 0; d < 4; d++) {
                int nx = x + dirX[d];
                int ny = y + dirY[d];
                
                if (nx >= 0 && nx < m && ny >= 0 && ny < n && classroom[nx].charAt(ny) != 'X') {
                    int nmask = mask;
                    int ne = e - 1; 
                    
                    if (ne < 0) continue; 
                    
                    char nc = classroom[nx].charAt(ny);
                    if (nc == 'L') {
                        nmask |= (1 << litterIdx[nx][ny]);
                    } else if (nc == 'R') {
                        ne = energy; 
                    }
                    
                    if (nmask == fullMask) {
                        return steps + 1;
                    }
                    
                    if (ne > bestEnergy[nx][ny][nmask]) {
                        bestEnergy[nx][ny][nmask] = ne;
                        q.add(new int[]{nx, ny, nmask, ne, steps + 1});
                    }
                }
            }
        }
        
        return -1; 
    }
}
