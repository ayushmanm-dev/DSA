class Solution {
    public long sumAndMultiply(int n) {
        if (n == 0) return 0;
        
        String s = String.valueOf(n);
        StringBuilder sb = new StringBuilder();
        long sum = 0;
        
        for (char c : s.toCharArray()) {
            if (c != '0') {
                sb.append(c);
                sum += (c - '0');
            }
        }
        
        if (sb.length() == 0) return 0;
        
        long x = Long.parseLong(sb.toString());
        return x * sum;
    }
}
