import java.util.*;
class Solution {
    public long weightedSum(int[] parent, int[] nums) {
        int []malviretho =parent;
        int n = parent.length;
        int[]depth =new int[n];
        List<List<Integer>> children = new ArrayList<>();
        for (int i =0;i<n;i++){
            children.add(new ArrayList<>());
        }
        for (int i=1;i<n;i++){
            children.get(parent[i]).add(i);
        }
        Queue<Integer>queue =new LinkedList<>();
        queue.offer(0);
        depth[0]=1;
        int h =1;
        while(!queue.isEmpty()){
            int u =queue.poll();
            h =Math.max(h,depth[u]);
            for (int v : children.get(u)){
                depth[v]=depth[u]+1;
                queue.offer(v);
            }
        }
        long totalWeight =0;
        for (int i = 0;i<n;i++){
            totalWeight += (long) nums[i]*(h-depth[i]+1);
        }
        return totalWeight;
    }
