class Solution {
    public int reverseDegree(String s) {
        int totalDegree = 0;
        
        for (int i = 0; i < s.length(); i++) {
            int stringPosition = i + 1;
            
            
            int reversedAlphabetPosition = 26 - (s.charAt(i) - 'a');
            
            totalDegree += reversedAlphabetPosition * stringPosition;
        }
        
        return totalDegree;
    }
}
