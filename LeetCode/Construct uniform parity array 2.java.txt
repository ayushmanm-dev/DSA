class Solution {
    public boolean uniformArray(int[] nums1) {
        int min = Integer.MAX_VALUE;
        boolean hasOdd = false;
        boolean hasEven = false;
        
        for (int num : nums1) {
            if (num < min) {
                min = num;
            }
            if (num % 2 == 0) {
                hasEven = true;
            } else {
                hasOdd = true;
            }
        }
        
        if (!hasOdd || !hasEven) {
            return true;
        }
        
        return min % 2 != 0;
    }
}
