class Solution {
    public String[] createGrid(int m, int n) {
        StringBuilder[] gridRows = new StringBuilder[m];
        
        for (int i = 0; i < m; i++) {
            StringBuilder sb = new StringBuilder();
            for (int j = 0; j < n; j++) {
                sb.append('#');
            }
            gridRows[i] = sb;
        }
        
        for (int j = 0; j < n; j++) {
            gridRows[0].setCharAt(j, '.');
        }
        
        for (int i = 0; i < m; i++) {
            gridRows[i].setCharAt(n - 1, '.');
        }
        
        String[] result = new String[m];
        for (int i = 0; i < m; i++) {
            result[i] = gridRows[i].toString();
        }
        
        return result;
    }
}
