import java.util.*;

class Solution {

    static class Node {
        long score;
        List<Integer> ids;

        Node(long score, List<Integer> ids) {
            this.score = score;
            this.ids = ids;
        }
    }

    public int[] maximumWeight(List<List<Integer>> intervals) {
        int n = intervals.size();

        int[][] arr = new int[n][4];

        for (int i = 0; i < n; i++) {
            arr[i][0] = intervals.get(i).get(0);
            arr[i][1] = intervals.get(i).get(1);
            arr[i][2] = intervals.get(i).get(2);
            arr[i][3] = i;
        }

        Arrays.sort(arr, (a, b) -> {
            if (a[0] != b[0])
                return Integer.compare(a[0], b[0]);

            return Integer.compare(a[3], b[3]);
        });

        int[] left = new int[n];

        for (int i = 0; i < n; i++) {
            left[i] = arr[i][0];
        }

        int[] next = new int[n];

        for (int i = 0; i < n; i++) {
            next[i] = upperBound(left, arr[i][1]);
        }

        Node[][] dp = new Node[n + 1][5];

        for (int k = 0; k <= 4; k++) {
            dp[n][k] = new Node(0, new ArrayList<>());
        }

        for (int i = n - 1; i >= 0; i--) {

            dp[i][0] = new Node(0, new ArrayList<>());

            for (int k = 1; k <= 4; k++) {

                Node skip = dp[i + 1][k];

                Node nextNode = dp[next[i]][k - 1];

                List<Integer> takeIds = new ArrayList<>(nextNode.ids);
                takeIds.add(arr[i][3]);

                Collections.sort(takeIds);

                Node take = new Node(
                    nextNode.score + arr[i][2],
                    takeIds
                );

                dp[i][k] = better(skip, take);
            }
        }

        List<Integer> answer = dp[0][4].ids;

        int[] result = new int[answer.size()];

        for (int i = 0; i < answer.size(); i++) {
            result[i] = answer.get(i);
        }

        return result;
    }

    private int upperBound(int[] arr, int target) {
        int lo = 0;
        int hi = arr.length;

        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;

            if (arr[mid] <= target) {
                lo = mid + 1;
            } else {
                hi = mid;
            }
        }

        return lo;
    }

    private Node better(Node a, Node b) {

        if (a.score != b.score) {
            return a.score > b.score ? a : b;
        }

        int minSize = Math.min(a.ids.size(), b.ids.size());

        for (int i = 0; i < minSize; i++) {
            if (!a.ids.get(i).equals(b.ids.get(i))) {
                return a.ids.get(i) < b.ids.get(i) ? a : b;
            }
        }

        return a.ids.size() <= b.ids.size() ? a : b;
    }
}
