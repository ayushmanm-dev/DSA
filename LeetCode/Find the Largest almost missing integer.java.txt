class Solution {
    public int largestInteger(int[] nums, int k) {
        int n = nums.length;
        int maxVal = -1;
        
        if (k == n) {
            for (int num : nums) {
                maxVal = Math.max(maxVal, num);
            }
            return maxVal;
        }
        
        int[] freq = new int[51];
        for (int num : nums) {
            freq[num]++;
        }
        
        if (k == 1) {
            for (int i = 0; i <= 50; i++) {
                if (freq[i] == 1) {
                    maxVal = Math.max(maxVal, i);
                }
            }
            return maxVal;
        }
        
        int first = nums[0];
        int last = nums[n - 1];
        
        if (freq[first] == 1) {
            maxVal = Math.max(maxVal, first);
        }
        if (freq[last] == 1) {
            maxVal = Math.max(maxVal, last);
        }
        
        return maxVal;
    }
}
