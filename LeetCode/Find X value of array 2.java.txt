class Solution {
    
    class Node {
        int prod;
        int[] freq;
        
        Node(int k) {
            freq = new int[k];
            prod = 1; 
        }
    }
    
    Node[] tree;
    int k_val;
    
    private Node merge(Node left, Node right) {
        Node res = new Node(k_val);
        res.prod = (left.prod * right.prod) % k_val;
        
        for (int i = 0; i < k_val; i++) {
            res.freq[i] += left.freq[i];
        }
        
        for (int i = 0; i < k_val; i++) {
            if (right.freq[i] > 0) {
                int new_mod = (left.prod * i) % k_val;
                res.freq[new_mod] += right.freq[i];
            }
        }
        
        return res;
    }
    
    private void build(int node, int start, int end, int[] nums) {
        if (start == end) {
            tree[node] = new Node(k_val);
            tree[node].prod = nums[start] % k_val;
            tree[node].freq[nums[start] % k_val] = 1;
            return;
        }
        int mid = start + (end - start) / 2;
        build(2 * node, start, mid, nums);
        build(2 * node + 1, mid + 1, end, nums);
        
        tree[node] = merge(tree[2 * node], tree[2 * node + 1]);
    }
    
    private void update(int node, int start, int end, int idx, int val) {
        if (start == end) {
            tree[node].prod = val % k_val;
            for (int i = 0; i < k_val; i++) tree[node].freq[i] = 0; 
            tree[node].freq[val % k_val] = 1;
            return;
        }
        int mid = start + (end - start) / 2;
        if (idx <= mid) {
            update(2 * node, start, mid, idx, val);
        } else {
            update(2 * node + 1, mid + 1, end, idx, val);
        }
        tree[node] = merge(tree[2 * node], tree[2 * node + 1]);
    }
    
    private Node query(int node, int start, int end, int l, int r) {
        if (r < start || end < l) {
            return new Node(k_val); 
        }
        if (l <= start && end <= r) {
            return tree[node];
        }
        int mid = start + (end - start) / 2;
        Node left = query(2 * node, start, mid, l, r);
        Node right = query(2 * node + 1, mid + 1, end, l, r);
        
        return merge(left, right);
    }

    public int[] resultArray(int[] nums, int k, int[][] queries) {
        int n = nums.length;
        this.k_val = k;
        this.tree = new Node[4 * n];
        
        build(1, 0, n - 1, nums);
        
        int[] result = new int[queries.length];
        
        for (int i = 0; i < queries.length; i++) {
            int idx = queries[i][0];
            int val = queries[i][1];
            int start = queries[i][2];
            int x = queries[i][3];
            
            update(1, 0, n - 1, idx, val);
            
            Node resNode = query(1, 0, n - 1, start, n - 1);
            
            result[i] = resNode.freq[x];
        }
        
        return result;
    }
}
