class Solution {

    public int[] pivotArray(int[] num1, int pivot) {

        int n = num1.length;
        int[] result = new int[n];
        int index = 0;
        for (int num : num1) {
            if (num < pivot) {
                result[index++] = num;
            }
        }
        for (int num : num1) {
            if (num == pivot) {
                result[index++] = num;
            }
        }
        for (int num : num1) {
            if (num > pivot) {
                result[index++] = num;
            }
        }

        return result;
    }
}
