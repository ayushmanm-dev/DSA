class Solution {
    public long[] resultArray(int[] nums, int k) {
        long[] result = new long[k];
        
        long[] prev = new long[k];
        int[] active = new int[k];
        int activeCount = 0;

        for (int num : nums) {
            long[] curr = new long[k];
            int[] nextActive = new int[k];
            int nextActiveCount = 0;
            boolean[] seen = new boolean[k];

            int rem = num % k;
            curr[rem]++;
            seen[rem] = true;
            nextActive[nextActiveCount++] = rem;

            for (int i = 0; i < activeCount; i++) {
                int r = active[i];
                int newRem = (int) (((long) r * num) % k);
                
                curr[newRem] += prev[r];
                
                if (!seen[newRem]) {
                    seen[newRem] = true;
                    nextActive[nextActiveCount++] = newRem;
                }
            }

            for (int i = 0; i < nextActiveCount; i++) {
                int r = nextActive[i];
                result[r] += curr[r];
            }

            prev = curr;
            active = nextActive;
            activeCount = nextActiveCount;
        }

        return result;
    }
}
