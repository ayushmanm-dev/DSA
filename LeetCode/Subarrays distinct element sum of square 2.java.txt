class Solution {
    public int sumCounts(int[] nums) {
        int n = nums.length;
        long MOD = 1_000_000_007;

        Map<Integer, Integer> lastOccurrence = new HashMap<>();
        
        SegmentTree tree = new SegmentTree(n);
        long totalSum = 0;
        long currentTotalSquares = 0;

        for (int j = 0; j < n; j++) {
            int val = nums[j];
            int p = lastOccurrence.getOrDefault(val, -1);
            
            if (p + 1 <= j) {
                tree.update(1, 0, n - 1, p + 1, j);
            }
            
            lastOccurrence.put(val, j);
            
            currentTotalSquares = tree.query(1, 0, n - 1, 0, j);
            totalSum = (totalSum + currentTotalSquares) % MOD;
        }

        return (int) totalSum;
    }

    static class SegmentTree {
        int n;
        long[] sumSq;
        long[] sumLin;
        long[] lazy;
        long MOD = 1_000_000_007;

        public SegmentTree(int n) {
            this.n = n;
            sumSq = new long[4 * n];
            sumLin = new long[4 * n];
            lazy = new long[4 * n];
        }

        private void apply(int node, int l, int r, long val) {
            long len = (r - l + 1);
            sumSq[node] = (sumSq[node] + 2 * val % MOD * sumLin[node] % MOD + val * val % MOD * len) % MOD;
            sumLin[node] = (sumLin[node] + val * len) % MOD;
            lazy[node] = (lazy[node] + val) % MOD;
        }

        private void push(int node, int l, int r) {
            if (lazy[node] != 0) {
                int mid = (l + r) / 2;
                apply(2 * node, l, mid, lazy[node]);
                apply(2 * node + 1, mid + 1, r, lazy[node]);
                lazy[node] = 0;
            }
        }

        public void update(int node, int l, int r, int ql, int qr) {
            if (ql <= l && r <= qr) {
                apply(node, l, r, 1);
                return;
            }
            push(node, l, r);
            int mid = (l + r) / 2;
            if (ql <= mid) update(2 * node, l, mid, ql, qr);
            if (qr > mid) update(2 * node + 1, mid + 1, r, ql, qr);
            
            sumSq[node] = (sumSq[2 * node] + sumSq[2 * node + 1]) % MOD;
            sumLin[node] = (sumLin[2 * node] + sumLin[2 * node + 1]) % MOD;
        }

        public long query(int node, int l, int r, int ql, int qr) {
            if (ql <= l && r <= qr) {
                return sumSq[node];
            }
            push(node, l, r);
            int mid = (l + r) / 2;
            long res = 0;
            if (ql <= mid) res = (res + query(2 * node, l, mid, ql, qr)) % MOD;
            if (qr > mid) res = (res + query(2 * node + 1, mid + 1, r, ql, qr)) % MOD;
            return res;
        }
    }
}
