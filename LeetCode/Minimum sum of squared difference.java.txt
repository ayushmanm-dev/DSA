import java.util.Arrays;

class Solution {
    public long minSumSquareDiff(int[] nums1, int[] nums2, int k1, int k2) {
        int n = nums1.length;
        long totalK = (long) k1 + k2;
        
        int maxDiff = 0;
        int[] diffCounts = new int[100001]; 
        
        for (int i = 0; i < n; i++) {
            int d = Math.abs(nums1[i] - nums2[i]);
            diffCounts[d]++;
            if (d > maxDiff) {
                maxDiff = d;
            }
        }
        
        for (int d = maxDiff; d > 0 && totalK > 0; d--) {
            if (diffCounts[d] == 0) continue;
            
            long operationsNeeded = diffCounts[d];
            long operationsToApply = Math.min(totalK, operationsNeeded);
            
            diffCounts[d] -= operationsToApply;
            diffCounts[d - 1] += operationsToApply;
            totalK -= operationsToApply;
        }
        
        long result = 0;
        for (int d = 1; d <= maxDiff; d++) {
            if (diffCounts[d] > 0) {
                result += (long) diffCounts[d] * d * d;
            }
        }
        
        return result;
    }
}
